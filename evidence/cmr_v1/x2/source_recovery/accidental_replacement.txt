def write_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + "\n", encoding="utf-8")


def array_sha256(array):
    return hashlib.sha256(np.ascontiguousarray(array).tobytes()).hexdigest()


def source_hashes():
    return {p: file_sha256(ROOT / p) for p in SOURCE_PATHS}


def phase1_authority():
    """Import the existing loader/builder unchanged, without running its CLI."""
    path = ROOT / "tools/prepare_phase1_feature_bank.py"
    spec = importlib.util.spec_from_file_location("cmr_phase1_authority", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


class LayerExtractor:
    """Capture actual post-stage outputs or post-block CLS before final LN."""

    def __init__(self, model, encoder):
        self.model, self.encoder = model, encoder
        self.outputs = {}
        self.handles = []
        for layer in X2_ENCODER_LAYERS[encoder]:
            module = getattr(model, layer) if encoder.startswith("resnet") else model.encoder.layers[int(layer[1:]) - 1]
            self.handles.append(module.register_forward_hook(self._hook(layer)))

    def _hook(self, layer):
        def capture(module, inputs, output):
            self.outputs[layer] = output
        return capture

    def __call__(self, images):
        import torch
        self.outputs.clear()
        with torch.inference_mode():
            self.model(images)
            out = {}
            for layer in X2_ENCODER_LAYERS[self.encoder]:
                z = self.outputs[layer]
                if self.encoder.startswith("resnet"):
                    z = torch.nn.functional.adaptive_avg_pool2d(z, (1, 1)).flatten(1)
                else:
                    z = z[:, 0, :]
                out[layer] = z.detach().cpu().numpy().astype(np.float64)
        return out

    def close(self):
        for handle in self.handles:
            handle.remove()
        self.handles.clear()
        self.outputs.clear()


def freeze_maps(head):
    """Persist once; any later load validates and uses the frozen arrays."""
    dims = {"resnet18_imagenet1k_v1": (64, 128, 256, 512),
            "vit_b_16_imagenet1k_v1": (768, 768, 768, 768)}
    if MAP_MANIFEST.exists():
        manifest = json.loads(MAP_MANIFEST.read_text())
        if manifest["generation_code_sha256"] != source_hashes():
            raise RuntimeError("RFF generation implementation changed after map freeze")
    else:
        entries = []
        for encoder, layers in X2_ENCODER_LAYERS.items():
            for layer, dim in zip(layers, dims[encoder]):
                w, b = generate_rff(encoder, layer, dim)
                path = RUN / "feature_maps" / f"{encoder}__{layer}.npz"
                if path.exists():
                    # A interrupted first freeze must never overwrite its arrays.
                    with np.load(path, allow_pickle=False) as old:
                        if not np.array_equal(old["W"], w) or not np.array_equal(old["b"], b):
                            raise RuntimeError("preexisting RFF arrays disagree with frozen construction")
                else:
                    path.parent.mkdir(parents=True, exist_ok=True)
                    np.savez(path, W=w, b=b)
                entries.append({"encoder": encoder, "layer": layer, "input_dimension": dim,
                    "output_dimension": 256, "sigma": 1.0, "derived_seed": rff_seed(encoder, layer),
                    "W_sha256": array_sha256(w), "b_sha256": array_sha256(b), "dtype": "float64",
                    "path": path.relative_to(ROOT).as_posix(), "artifact_sha256": file_sha256(path),
                    "git_head": head})
        manifest = {"manifest_type": "CMR_V1_FEATURE_MAP_MANIFEST", "status": "FROZEN",
                    "generation_code_sha256": source_hashes(), "git_head": head, "maps": entries}
        write_json(MAP_MANIFEST, manifest)
    expected = {(e, l) for e, layers in X2_ENCODER_LAYERS.items() for l in layers}
    if manifest.get("status") != "FROZEN" or len(manifest["maps"]) != 8 or {(e["encoder"], e["layer"]) for e in manifest["maps"]} != expected:
        raise RuntimeError("invalid frozen feature-map manifest")
    maps = {}
    for entry in manifest["maps"]:
        path = ROOT / entry["path"]
        if file_sha256(path) != entry["artifact_sha256"]:
            raise RuntimeError("frozen RFF artifact hash mismatch")
        with np.load(path, allow_pickle=False) as z:
            w, b = z["W"].copy(), z["b"].copy()
        if w.dtype != np.float64 or b.dtype != np.float64 or array_sha256(w) != entry["W_sha256"] or array_sha256(b) != entry["b_sha256"]:
            raise RuntimeError("frozen RFF array hash/dtype mismatch")
        maps[(entry["encoder"], entry["layer"])] = (w, b, entry)
    return maps


def sample_identity(dataset, dataset_id, split, raw_root):
    """Canonical row IDs are ordered exactly as the unchanged loader."""
    from torch.utils.data import Subset
    if isinstance(dataset, Subset):
        ids = [Path(dataset.dataset.samples[i][0]).relative_to(raw_root).as_posix() for i in dataset.indices]
    elif hasattr(dataset, "filenames"):
        ids = [f"{dataset_id}/{split}/{i}:{name}" for i, name in enumerate(dataset.filenames)]
    else:
        ids = [f"{dataset_id}/{split}/{i}" for i in range(len(dataset))]
    return ids, hashlib.sha256(("\n".join(ids) + "\n").encode()).hexdigest()


def raw_dataset_hashes(dataset_id, raw_root, train_dataset):
    if dataset_id == "cifar100":
        paths = sorted((raw_root / "cifar-100-python").glob("*"))
    elif dataset_id in {"pathmnist", "dermamnist"}:
        paths = [raw_root / f"{dataset_id}.npz"]
    else:
        paths = [Path(p) for p, _ in train_dataset.dataset.samples]
    h = hashlib.sha256()
    count = 0
    for path in paths:
        if path.is_file():
            h.update(path.relative_to(raw_root).as_posix().encode())
            h.update(bytes.fromhex(file_sha256(path)))
            count += 1
    if not count:
        raise RuntimeError("raw dataset provenance is missing")
    return {"raw_file_count": count, "raw_content_manifest_sha256": h.hexdigest()}


def preflight(head, remote):
    import yaml
    if head != remote or head != EXPECTED_HEAD:
        raise RuntimeError(f"baseline mismatch: local={head} remote={remote} expected={EXPECTED_HEAD}")
    frozen = json.loads((ROOT / "docs/phases/cmr_v1/CMR_V1_FREEZE_MANIFEST.json").read_text())
    aliases = {"CROSS_MECHANISM_REPLICATION_PROTOCOL_V1.0.md": AUTHORITY_PATHS[0],
               "cross_mechanism_replication_protocol_v1.yaml": AUTHORITY_PATHS[1],
               "CMR_V1_GATE_MATRIX.csv": AUTHORITY_PATHS[2]}
    for name, path in aliases.items():
        if file_sha256(ROOT / path) != frozen["files"][name]["sha256"]:
            raise RuntimeError("scientific authority changed")
    protocol = yaml.safe_load((ROOT / AUTHORITY_PATHS[1]).read_text())
    if protocol["protocol"] != yaml.safe_load((ROOT / "evidence/cmr_v1/x1/protocol_snapshot.yaml").read_text())["protocol"]:
        raise RuntimeError("protocol identity changed since X1")
    gate = json.loads((ROOT / "evidence/cmr_v1/x1/gate_summary.json").read_text())
    if gate["decision"] != "X1_PASS_EXACT_KERNEL_PROTOTYPE_LAYER_COLLISION" or not gate["all_gates_pass"]:
        raise RuntimeError("X1 not PASS")
    evidence = json.loads((ROOT / "evidence/cmr_v1/x1/evidence_manifest.json").read_text())
    for entry in evidence["artifacts"]:
        if file_sha256(ROOT / entry["path"]) != entry["sha256"]:
            raise RuntimeError("X1 evidence artifact changed")
    # The X1 sources remain verifiable in their immutable Git commit, although
    # the authorized X2 implementation extends two of them in the working tree.
    for entry in evidence["source_files"]:
        data = subprocess.check_output(["git", "show", f"HEAD:{entry['path']}"], cwd=ROOT)
        if hashlib.sha256(data).hexdigest() != entry["sha256"]:
            raise RuntimeError("X1 code was not committed intact")
    for root in (ROOT / "runs/cmr_v1", ROOT / "evidence/cmr_v1"):
        if any((root / f"x{i}").exists() for i in (3, 4, 5)):
            raise RuntimeError("unauthorized later-stage artifacts exist")
    return protocol


def extract_banks(raw_root, device, batch_size, workers, head, maps):
    import torch
    import torchvision
    import medmnist
    from torch.utils.data import DataLoader
    authority = phase1_authority()
    phase1 = json.loads((ROOT / "configs/phase1_feature_manifest.json").read_text())
    old = {b["bank_id"]: b for b in phase1["banks"]}
    entries = []
    for encoder, layers in X2_ENCODER_LAYERS.items():
        model, weights, tv_version, torch_version = authority._build_encoder(encoder, device)
        state_sha = authority._state_dict_sha256(model)
        expected_state = old[f"cifar100__{encoder}"]["encoder_state_sha256"]
        if state_sha != expected_state or tv_version != old[f"cifar100__{encoder}"]["torchvision_version"]:
            raise RuntimeError("encoder state or torchvision differs from Phase-1 authority")
        transform = weights.transforms()
        transform_id = f"torchvision_weights_transforms::{weights}"
        transform_sha = hashlib.sha256(repr(transform).encode()).hexdigest()
        write_json(RUN / f"encoder_preflight_{encoder}.json", {
            "encoder": encoder, "state_sha256": state_sha, "preprocessing_sha256": transform_sha,
            "preprocessing_repr": repr(transform), "weights_enum": str(weights), "git_head": head})
        extractor = LayerExtractor(model, encoder)
        try:
            for dataset_id in X2_DATASETS:
                bank_id = f"{dataset_id}__{encoder}"
                previous = old[bank_id]
                tr, te, split_meta = authority._dataset(dataset_id, raw_root, transform, False, 20260908)
                if len(tr) != previous["train_n"] or len(te) != previous["test_n"] or split_meta["split_id"] != previous["split_id"]:
                    raise RuntimeError("dataset split/count differs from Phase-1")
                for key in ("train_indices_sha256", "test_indices_sha256", "medmnist_version"):
                    if key in previous and split_meta.get(key) != previous[key]:
                        raise RuntimeError(f"Phase-1 dataset authority mismatch: {key}")
                raw_provenance = raw_dataset_hashes(dataset_id, raw_root, tr)
                for split, dataset in (("train", tr), ("test", te)):
                    ids, id_sha = sample_identity(dataset, dataset_id, split, raw_root)
                    ids_path = RUN / "feature_banks" / f"{bank_id}__{split}__sample_ids.txt"
                    ids_path.parent.mkdir(parents=True, exist_ok=True)
                    ids_path.write_text("\n".join(ids) + "\n", encoding="utf-8")
                    loader = DataLoader(dataset, batch_size=batch_size, shuffle=False, num_workers=workers,
                        pin_memory=device.startswith("cuda"), collate_fn=collate_medmnist if dataset_id in {"pathmnist", "dermamnist"} else None)
                    n = len(dataset)
                    feature_arrays = {layer: np.lib.format.open_memmap(
                        RUN / "feature_banks" / f"{bank_id}__{split}__{layer}.npy", mode="w+", dtype=np.float64, shape=(n, 256)) for layer in layers}
                    labels = np.empty(n, dtype=np.int64)
                    offset = 0
                    for images, y in loader:
                        features = extractor(images.to(device, non_blocking=True))
                        size = len(images)
                        labels[offset:offset + size] = np.asarray(y).reshape(-1)
                        for layer in layers:
                            w, b, _ = maps[(encoder, layer)]
                            feature_arrays[layer][offset:offset + size] = apply_rff(normalize_layer_features(features[layer]), w, b)
                        offset += size
                        if offset % (batch_size * 100) == 0 or offset == n:
                            print(f"[CMR X2 feature extraction] {bank_id} {split} {offset}/{n}", flush=True)
                    if offset != n or not np.array_equal(np.unique(labels), np.arange(previous["n_classes"])):
                        raise RuntimeError("feature-bank row/class coverage mismatch")
                    label_path = RUN / "feature_banks" / f"{bank_id}__{split}__labels.npy"
                    np.save(label_path, labels)
                    for layer in layers:
                        x = feature_arrays[layer]
                        x.flush()
                        if not np.isfinite(x).all():
                            raise RuntimeError("non-finite feature-bank values")
                        path = RUN / "feature_banks" / f"{bank_id}__{split}__{layer}.npy"
                        entry = {"dataset": dataset_id, "source_authority": "tools/prepare_phase1_feature_bank.py::_dataset",
                            "phase1_manifest_bank": previous, "split": split, "split_semantics": split_meta,
                            "record_count": n, "class_count": previous["n_classes"],
                            "canonical_sample_ids_sha256": id_sha, "sample_ids_path": ids_path.relative_to(ROOT).as_posix(),
                            "sample_ids_file_sha256": file_sha256(ids_path), "encoder": encoder, "pretrained_weights_enum": str(weights),
                            "encoder_state_sha256": state_sha, "preprocessing_identity": transform_id,
                            "preprocessing_sha256": transform_sha, "layer": layer,
                            "raw_layer_dimension": maps[(encoder, layer)][2]["input_dimension"], "l2_normalization": True,
                            "rff_map_sha256": maps[(encoder, layer)][2]["artifact_sha256"], "final_rff_dimension": 256,
                            "dtype": "float64", "source_code_sha256": source_hashes(), "git_head": head,
                            "torch_version": torch_version, "torchvision_version": tv_version,
                            "medmnist_version": medmnist.__version__ if dataset_id in {"pathmnist", "dermamnist"} else None,
                            "path": path.relative_to(ROOT).as_posix(), "sha256": file_sha256(path),
                            "labels_path": label_path.relative_to(ROOT).as_posix(), "labels_sha256": file_sha256(label_path),
                            **raw_provenance}
                        entries.append(entry)
                    del feature_arrays
        finally:
            extractor.close()
            del model, extractor
            if device.startswith("cuda"):
                torch.cuda.empty_cache()
    manifest = {"manifest_type": "CMR_V1_FEATURE_BANK_MANIFEST", "status": "FROZEN",
        "git_head": head, "feature_map_manifest_sha256": file_sha256(MAP_MANIFEST),
        "source_code_sha256": source_hashes(), "banks": entries}
    write_json(BANK_MANIFEST, manifest)
    return manifest


def validate_bank_manifest(manifest):
    expected = {(d, e, l, s) for d in X2_DATASETS for e, layers in X2_ENCODER_LAYERS.items()
                for l in layers for s in ("train", "test")}
    actual = {(b["dataset"], b["encoder"], b["layer"], b["split"]) for b in manifest["banks"]}
    if actual != expected or len(manifest["banks"]) != 64 or manifest["status"] != "FROZEN":
        raise RuntimeError("feature-bank manifest lacks frozen coverage")
    if manifest["source_code_sha256"] != source_hashes() or manifest["feature_map_manifest_sha256"] != file_sha256(MAP_MANIFEST):
        raise RuntimeError("feature-bank source/map provenance changed")
    for bank in manifest["banks"]:
        for path_key, hash_key in (("path", "sha256"), ("labels_path", "labels_sha256"), ("sample_ids_path", "sample_ids_file_sha256")):
            if file_sha256(ROOT / bank[path_key]) != bank[hash_key]:
                raise RuntimeError("frozen feature-bank artifact hash mismatch")


def evaluate_banks(manifest):
    banks = {(b["dataset"], b["encoder"], b["layer"], b["split"]): b for b in manifest["banks"]}
    conditions = []
    for dataset in X2_DATASETS:
        for encoder, layers in X2_ENCODER_LAYERS.items():
            bacc, audits = {}, []
            for layer in layers:
                train = banks[(dataset, encoder, layer, "train")]
                test = banks[(dataset, encoder, layer, "test")]
                xtr = np.load(ROOT / train["path"], mmap_mode="r", allow_pickle=False)
                ytr = np.load(ROOT / train["labels_path"], allow_pickle=False)
                xte = np.load(ROOT / test["path"], mmap_mode="r", allow_pickle=False)
                yte = np.load(ROOT / test["labels_path"], allow_pickle=False)
                clients = dirichlet_label_partition(ytr, 20, 0.10, 20260908, bank_id=f"{dataset}__{encoder}")
                message, audit = aggregate_client_messages(xtr, ytr, clients)
                prototypes = reconstruct_prototypes(message)
                predictions = predict_prototype_batch(xte, prototypes)
                bacc[layer] = bacc_from_predictions(yte, predictions)
                audits.append(audit)
                write_json(RUN / "messages" / f"{dataset}__{encoder}__{layer}.json", {
                    "communication": {str(c): {"count": v["count"], "vector_sum": v["vector_sum"].tolist()} for c, v in message.items()},
                    "recovery": audit, "client_assignment_sha256": array_sha256(clients)})
            recovery = {"max_abs": max(a["max_abs"] for a in audits), "passed": all(a["passed"] for a in audits)}
            condition = summarize_x2_condition(dataset, encoder, bacc, recovery)
            conditions.append(condition)
            write_json(RUN / "partial_condition_results.json", {"conditions": conditions})
            print(f"[CMR X2 outcome] {dataset} {encoder}: {bacc}", flush=True)
    return conditions


def run_x2(raw_root=None, device=None, batch_size=64, workers=4):
    """Single formal run, with explicit outcome access marker and failures saved."""
    import torch
    import torchvision
    run_id = len(list(RUN.glob("engineering_failure_*.json"))) + 1 if RUN.exists() else 1
    if (RUN / "run_summary.json").exists() or (RUN / "outcome_access.json").exists():
        print("[CMR X2] Existing formal outcome artifacts: automatic rerun forbidden.", file=sys.stderr)
        return 2
    try:
        head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
        remote = subprocess.check_output(["git", "ls-remote", "origin", "refs/heads/main"], cwd=ROOT, text=True).split()[0]
        protocol = preflight(head, remote)
        raw_root = Path(raw_root or os.environ.get("CMR_PHASE1_RAW_ROOT", str(ROOT / "data/phase1_raw"))).resolve()
        if not raw_root.is_dir():
            raise FileNotFoundError(f"Phase-1 raw dataset root missing: {raw_root}")
        targets = ["tests/test_cmr_v1_x1.py", "tests/test_cmr_v1_protocol.py", "tests/test_cmr_v1_x2.py"]
        env = os.environ.copy()
        env["PYTHONDONTWRITEBYTECODE"] = "1"
        # Protocol T11 is a pre-X2 entry test. Once maps freeze, later recovery
        # must validate X2 maps instead of rerunning the historical X1 boundary.
        if MAP_MANIFEST.exists() or BANK_MANIFEST.exists():
            targets.remove("tests/test_cmr_v1_protocol.py")
        tests = subprocess.run([sys.executable, "-B", "-m", "pytest", *targets, "-q", "-p", "no:cacheprovider"], cwd=ROOT, env=env, capture_output=True, text=True)
        if tests.returncode:
            raise RuntimeError("pre-outcome tests failed:\n" + tests.stdout + tests.stderr)
        device = device or ("cuda" if torch.cuda.is_available() else "cpu")
        torch.backends.cuda.matmul.allow_tf32 = False
        torch.backends.cudnn.allow_tf32 = False
        torch.backends.cudnn.benchmark = False
        torch.use_deterministic_algorithms(True)
        os.environ.setdefault("CUBLAS_WORKSPACE_CONFIG", ":4096:8")
        torch.set_num_threads(4)
        environment = {"python": platform.python_version(), "python_executable": sys.executable,
            "torch": torch.__version__, "torchvision": torchvision.__version__, "numpy": np.__version__,
            "device": device, "device_name": torch.cuda.get_device_name(0) if device.startswith("cuda") else "CPU",
            "tf32": False, "deterministic_algorithms": True, "batch_size": batch_size, "workers": workers}
        write_json(RUN / "preflight.json", {"head": head, "remote_main": remote, "protocol": protocol["protocol"],
            "environment": environment, "tests": {"targets": targets, "returncode": tests.returncode, "stdout": tests.stdout},
            "source_code_sha256": source_hashes(), "authority_sha256": {p: file_sha256(ROOT / p) for p in AUTHORITY_PATHS}})
        maps = freeze_maps(head)
        if BANK_MANIFEST.exists():
            manifest = json.loads(BANK_MANIFEST.read_text())
        else:
            manifest = extract_banks(raw_root, device, batch_size, workers, head, maps)
        validate_bank_manifest(manifest)
        write_json(RUN / "outcome_access.json", {"stage": "X2", "git_head": head,
            "feature_map_manifest_sha256": file_sha256(MAP_MANIFEST), "feature_bank_manifest_sha256": file_sha256(BANK_MANIFEST),
            "source_code_sha256": source_hashes(), "semantics": "all frozen feature banks validated before evaluating any BACC"})
        conditions = evaluate_banks(manifest)
        gate = evaluate_x2_gates(conditions)
        provenance = {"protocol_id": protocol["protocol"]["id"], "protocol_version": protocol["protocol"]["version"],
            "protocol_status": protocol["protocol"]["status"], "git_head_before_x2": head,
            "environment": environment, "source_code_sha256": source_hashes(),
            "authority_sha256": {p: file_sha256(ROOT / p) for p in AUTHORITY_PATHS},
            "feature_map_manifest_sha256": file_sha256(MAP_MANIFEST), "feature_bank_manifest_sha256": file_sha256(BANK_MANIFEST)}
        write_json(RUN / "run_summary.json", {**provenance, **gate, "conditions": conditions})
        csv_path = RUN / "condition_results.csv"
        fields = ["dataset", "encoder", "BACC_action_1", "BACC_action_2", "BACC_action_3", "BACC_action_4",
                  "best_layer", "second_best_layer", "best_margin", "layer_spread", "action_relevant", "unique_best", "aggregation_recovery_max_abs", "finite_pass"]
        with csv_path.open("w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=fields)
            writer.writeheader()
            for c in conditions:
                row = {k: c[k] for k in fields if k in c}
                row.update({f"BACC_action_{i + 1}": c["balanced_accuracy"][a] for i, a in enumerate(X2_ENCODER_LAYERS[c["encoder"]])})
                writer.writerow(row)
        write_json(EVIDENCE / "gate_summary.json", {**provenance, **gate})
        (EVIDENCE / "protocol_snapshot.yaml").write_bytes((ROOT / AUTHORITY_PATHS[1]).read_bytes())
        artifacts = [RUN / "run_summary.json", csv_path, RUN / "preflight.json", RUN / "outcome_access.json",
            EVIDENCE / "gate_summary.json", EVIDENCE / "protocol_snapshot.yaml", MAP_MANIFEST, BANK_MANIFEST]
        write_json(EVIDENCE / "evidence_manifest.json", {**provenance, "decision": gate["decision"],
            "artifacts": [{"path": p.relative_to(ROOT).as_posix(), "sha256": file_sha256(p), "bytes": p.stat().st_size} for p in artifacts]})
        print(json.dumps(gate, indent=2), flush=True)
        return 0 if gate["all_gates_pass"] else 1
    except Exception as exc:
        write_json(RUN / f"engineering_failure_{run_id:03d}.json", {"stage": "X2", "decision": "CMR_INTEGRITY_BLOCKED",
            "error": str(exc), "traceback": traceback.format_exc(), "outcome_access_occurred": (RUN / "outcome_access.json").exists()})
        print(f"[CMR X2] Engineering failure preserved: {exc}", file=sys.stderr)
        return 2
